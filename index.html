Longest Substring Without Repeating Characters

Given a string, find the length of the longest substring without repeating characters.

function lengthOfLongestSubstring(s) { let seen = new Map(), start = 0, max = 0; for (let i = 0; i < s.length; i++) { if (seen.has(s[i]) && seen.get(s[i]) >= start) { start = seen.get(s[i]) + 1; } seen.set(s[i], i); max = Math.max(max, i - start + 1); } return max; }
